cmake_minimum_required(VERSION 3.21)
project(LumenDevices LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Build Lumen Windows devices with MSVC on Windows.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(SOFTCAM "${CMAKE_CURRENT_SOURCE_DIR}/vendor/softcam/src")
set(BASE_SOURCES amfilter amvideo combase dllentry dllsetup mtype source winutil wxdebug wxlist wxutil)
set(BASE_FILES "")
foreach(source IN LISTS BASE_SOURCES)
  list(APPEND BASE_FILES "${SOFTCAM}/baseclasses/${source}.cpp")
endforeach()
add_library(LumenCamera SHARED ${BASE_FILES}
  "${SOFTCAM}/softcam/softcam.cpp" "${SOFTCAM}/softcam/softcam.def"
  "${SOFTCAM}/softcamcore/DShowSoftcam.cpp" "${SOFTCAM}/softcamcore/FrameBuffer.cpp"
  "${SOFTCAM}/softcamcore/Misc.cpp" "${SOFTCAM}/softcamcore/SenderAPI.cpp" "${SOFTCAM}/softcamcore/Watchdog.cpp")
target_include_directories(LumenCamera PRIVATE "${SOFTCAM}" "${SOFTCAM}/baseclasses")
target_compile_definitions(LumenCamera PRIVATE WIN32 _CRT_SECURE_NO_WARNINGS)
target_link_libraries(LumenCamera PRIVATE strmiids winmm ole32 oleaut32 uuid advapi32)
add_executable(LumenDeviceBridge Bridge.cpp)
target_compile_definitions(LumenDeviceBridge PRIVATE UNICODE _UNICODE)
target_link_libraries(LumenDeviceBridge PRIVATE ole32 oleaut32 uuid avrt setupapi newdev advapi32)
add_executable(LumenLoopbackTest test/LoopbackTest.cpp)
target_compile_options(LumenLoopbackTest PRIVATE /UNDEBUG)
add_executable(LumenCameraTest test/CameraTest.cpp
  "${SOFTCAM}/softcamcore/FrameBuffer.cpp" "${SOFTCAM}/softcamcore/Misc.cpp" "${SOFTCAM}/softcamcore/Watchdog.cpp")
target_include_directories(LumenCameraTest PRIVATE "${SOFTCAM}")
target_compile_options(LumenCameraTest PRIVATE /UNDEBUG)
target_link_libraries(LumenCameraTest PRIVATE winmm)
enable_testing()
add_test(NAME Loopback COMMAND LumenLoopbackTest)
add_test(NAME Camera COMMAND LumenCameraTest)
install(TARGETS LumenCamera LumenDeviceBridge RUNTIME DESTINATION .)
